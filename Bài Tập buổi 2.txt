// Bài 4: Đếm số đơn hàng theo trạng thái
const orders = [
  { status: "pending" },
  { status: "completed" },
  { status: "pending" },
  { status: "cancelled" },
];

const countOrders = orders.reduce((sum, index) => {
  if (sum[index.status]) {
    sum[index.status] += 1;
  } else {
    sum[index.status] = 1;
  }
  return sum;
}, {});

console.log(countOrders);

const sales = [
  { user: "An", product: "Áo", price: 100, qty: 2 },
  { user: "Bình", product: "Giày", price: 300, qty: 1 },
  { user: "An", product: "Quần", price: 200, qty: 1 },
  { user: "Chi", product: "Áo", price: 100, qty: 5 },
];

const result = sales
  .filter((acc, index, arr) => {
    return index === arr.findIndex((p) => p.product === acc.product);
  })
  .map((p) => p.product);
console.log(result);

const totalQty = sales.reduce((sum, index) => {
  if (sum[index.product]) {
    sum[index.product] += index.qty;

  }

  return sum;
}, {});
