  const res = [];

// * Bài 2: Tìm tất cả các cặp số (a, b) sao cho a và b là số nguyên tố và a + b = n, với n là số nguyên dương cho trước.
function checknt(n) {
  if (n < 2) return false;

  for (let i = 2; i <= Math.sqrt(n); i++) {
    if (n % i === 0) {
      return false;
    }
  }

  return true;
}



let n = 20;

for (let i = 2; i <= n / 2; i++) {
  let j = n - i;
  if (checknt(i) && checknt(j) && i + j === n) {
    console.log(i, j);
  }
}

function checkchan(n) {
  let count = 0;
  while (n > 0) {
  
  count++
    n = Math.floor(n / 10);
  }

  return count % 2 === 0 ? true : false;
}
